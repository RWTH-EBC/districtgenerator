52def exemplary_plot(data): 

53 

54 # Sum heat demand of buildings 

55 heat = data.district[0]["user"].heat + data.district[1]["user"].heat 

56 # Unit conversion [kWh] 

57 heat = heat / (data.time["dataResolution"] / data.time["timeResolution"]) / 1000 

58 

59 # Create a dataframe that contains the timestamps 

60 date_range = pd.date_range(start='2023-01-01', periods=data.time["timeSteps"], freq='15min') 

61 df = pd.DataFrame(heat, index=date_range, columns=['Value']) 

62 

63 # Aggregate the data on a monthly basis (totalled value per month) 

64 monthly_data = df.resample('M').sum() 

65 

66 # Plot as bar chart 

67 plt.figure(figsize=(10, 6)) 

68 plt.bar(monthly_data.index.strftime('%b'), monthly_data['Value']) 

69 plt.ylabel('District space heat demand in kWh') 

70 plt.xticks(rotation=45) 

71 plt.tight_layout() 

72 plt.show()
