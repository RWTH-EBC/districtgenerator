326 def calc_annual_cost_device(self, dev, param, cap): 

327 """ 

328 Calculation of total investment costs including replacements (based on VDI 2067-1, pages 16-17). 

329 

330 Parameters 

331 ---------- 

332 dev : dictionary 

333 technology parameter 

334 param : dictionary 

335 economic parameters 

336 

337 Returns 

338 ------- 

339 annualized fix and variable investment 

340 """ 

341 

342 # Projektlaufzeit (observation_time) 

343 # Zinssatz (für Kapitalkosten) (interest_rate) 

344 # Anschaffungskosten inkl. Installation, etc. [€/kW] 

345 # Lebensdauer (life_time) [years] 

346 # Betriebs- und Wartungskosten (operation and maintenance costs, c_om) [% of Invest] 

347 

348 # Values from Technikkatalog (Langreder et al. 2024): 

349 # Langreder, Nora; Lettow, Frederik; Sahnoun, Malek; Kreidelmeyer, Sven; Wünsch, Aurel; Lengning, Saskia et al. 

350 # (2024): Technikkatalog Wärmeplanung. Hg. v. ifeu – Institut für Energie- und Umweltforschung Heidelberg, 

351 # Öko-Institut e.V., IER Stuttgart, adelphi consult GmbH, Becker Büttner Held PartGmbB, Prognos AG, et al. 

352 # Online available at: 

353 # https://api.kww-halle.de/fileadmin/user_upload/Technikkatalog_W%C3%A4rmeplanung_Version_1.1_August24.xlsx 

354 

355 observation_time = param["observation_time"] 

356 interest_rate = param["interest_rate"] 

357 q = 1 + param["interest_rate"] 

358 

359 # Calculate capital recovery factor 

360 CRF = ((q ** observation_time) * interest_rate) / ((q ** observation_time) - 1) 

361 

362 # Get device life time 

363 life_time = dev["life_time"] 

364 

365 # Number of required replacements 

366 n = int(math.floor(observation_time / life_time)) 

367 

368 # Investment for replacements 

369 invest_replacements = sum((q ** (-i * life_time)) for i in range(1, n + 1)) 

370 

371 # Residual value of final replacement 

372 res_value = ((n + 1) * life_time - observation_time) / life_time * (q ** (-observation_time)) 

373 

374 # Calculate annualized investments 

375 if life_time > observation_time: 

376 ann_factor = (1 - res_value) * CRF 

377 else: 

378 ann_factor = (1 + invest_replacements - res_value) * CRF 

379 

380 # Save capital recovery factor 

381 # param["CRF"] = CRF 

382 

383 # Total investment costs 

384 inv = dev["inv_var"] * cap 

385 # Annual investment costs 

386 c_inv= inv * ann_factor 

387 # Operation and maintenance costs 

388 c_om = dev["cost_om"] * inv 

389 # Total annual cost 

390 c_total = c_inv + c_om 

391 

392 return c_total
